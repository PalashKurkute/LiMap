### P1a — Vectorize `costmap_generator.py` (biggest win: 2863 ms → target <200 ms)

**File:** `core/planning/costmap_generator.py`  
**Root cause:** `generate_costmap()` has a pure Python `for c in active_cells:` loop (line 129) over up to ~40k structured-array rows. Every iteration does Python attribute access + arithmetic.

**Fix strategy — vectorize cost computation, keep painting loop:**

```python
def generate_costmap(self, grid, dynamic_tracks=None,
                     ignore_overhang_clearance=False,
                     enable_slope_compensation=True):
    costmap = np.zeros((self.ny, self.nx), dtype=np.uint8)
    active_cells = grid.get_active_cells()
    if len(active_cells) == 0:
        return costmap

    lattice = grid.lattice

    # --- Step 1: Vectorized coordinate computation ---
    ring_ids = active_cells["ring_id"].astype(np.int32)
    ring_sizes = np.array([r.cell_size for r in lattice.rings], dtype=np.float64)
    c_res = ring_sizes[ring_ids]  # per-cell resolution

    wx = (active_cells["ix"].astype(np.float64) + 0.5) * c_res
    wy = (active_cells["iy"].astype(np.float64) + 0.5) * c_res

    gx = ((wx - self.origin_x) / self.res).astype(np.int32)
    gy = ((wy - self.origin_y) / self.res).astype(np.int32)

    valid_mask = (gx >= 0) & (gx < self.nx) & (gy >= 0) & (gy < self.ny)

    # --- Step 2: Vectorized cost computation ---
    sem = active_cells["sem_id"].astype(np.int32)
    cnt = active_cells["count"].astype(np.int32)
    mean_z = active_cells["mean_z"].astype(np.float64)
    m2_z = active_cells["m2_z"].astype(np.float64)
    overhang_z = active_cells["overhang_z"].astype(np.float64)
    clearance = active_cells["clearance"].astype(np.float64)

    var_z = m2_z / np.maximum(cnt - 1, 1).astype(np.float64)

    # Vectorized semantic lookup (build lookup table once)
    sem_lut = np.zeros(256, dtype=np.int32)
    for k, v in SEMANTIC_COST_LOOKUP.items():
        sem_lut[k] = v
    base_cost = sem_lut[np.clip(sem, 0, 255)]

    rough_cost = np.minimum(var_z * self.roughness_weight, 80.0).astype(np.int32)
    uncertainty_cost = np.minimum(var_z * self.uncertainty_weight, 240.0).astype(np.int32)
    terrain_risk = np.maximum(rough_cost, uncertainty_cost)

    has_overhang = overhang_z < 900.0
    terrain_risk = np.where(has_overhang, 0, terrain_risk)

    # Simple obstacle detection (vectorized; PCA plane kept for enable_slope_compensation=True
    # but per-cell dot product also vectorized below)
    is_solid_obstacle = mean_z > -1.2  # fallback threshold

    if enable_slope_compensation:
        # fit_terrain_planes returns a dict keyed by bin (bx, by)
        patch_planes = self.fit_terrain_planes(grid)
        # Vectorized bin assignment
        bx = np.floor(wx / 6.0).astype(np.int32)
        by = np.floor(wy / 6.0).astype(np.int32)
        # Per-cell slope check (still a loop but over patch bins, not points)
        for i, (bxi, byi) in enumerate(zip(bx, by)):
            plane = patch_planes.get((int(bxi), int(byi)))
            if plane is not None and plane.is_traversable_grade:
                diff = np.array([wx[i] - plane.centroid[0],
                                 wy[i] - plane.centroid[1],
                                 mean_z[i] - plane.centroid[2]])
                is_solid_obstacle[i] = float(np.dot(diff, plane.normal)) > 0.3
    else:
        patch_planes = {}

    # Vectorized total cost
    if ignore_overhang_clearance:
        total_cost = np.where(
            (sem == 50) | has_overhang | is_solid_obstacle,
            COST_LETHAL,
            np.minimum(COST_LETHAL, base_cost + terrain_risk)
        ).astype(np.uint8)
    else:
        eff_clearance = np.where(
            (clearance > 900.0) & (mean_z > 0.3),
            mean_z - (-1.73),
            clearance
        )
        is_safe_underpass = (has_overhang | (mean_z > 0.3)) & (eff_clearance >= self.vehicle_height_m + 0.2)
        is_solid_lethal = ~is_safe_underpass & ((sem == 50) | is_solid_obstacle)

        total_cost = np.where(
            is_safe_underpass, terrain_risk,
            np.where(is_solid_lethal, COST_LETHAL,
                     np.minimum(COST_LETHAL, base_cost + terrain_risk))
        ).astype(np.uint8)

    # --- Step 3: Painting loop (only slice assignments, no arithmetic) ---
    half_w = np.maximum(1, np.round(c_res / (2.0 * self.res))).astype(np.int32)
    valid_idx = np.where(valid_mask)[0]
    for i in valid_idx:
        hw = int(half_w[i])
        y0 = max(0, int(gy[i]) - hw)
        y1 = min(self.ny, int(gy[i]) + hw + 1)
        x0 = max(0, int(gx[i]) - hw)
        x1 = min(self.nx, int(gx[i]) + hw + 1)
        costmap[y0:y1, x0:x1] = np.maximum(costmap[y0:y1, x0:x1], total_cost[i])

    # Dynamic track inflation (unchanged)
    if dynamic_tracks:
        for track in dynamic_tracks:
            cx_t, cy_t = track.state[0], track.state[1]
            cgx = int((cx_t - self.origin_x) / self.res)
            cgy = int((cy_t - self.origin_y) / self.res)
            radius_cells = int(max(track.bbox.length, track.bbox.width) / self.res) + 2
            y_min = max(0, cgy - radius_cells)
            y_max = min(self.ny, cgy + radius_cells + 1)
            x_min = max(0, cgx - radius_cells)
            x_max = min(self.nx, cgx + radius_cells + 1)
            costmap[y_min:y_max, x_min:x_max] = COST_LETHAL

    return costmap
```

**Key invariant:** The function's API signature, return type, and semantic behavior are **identical** to the original. Only the internal computation path changes.

### P1b — Numba JIT `_insert_batch` (1185 ms → target <50 ms warm)

**File:** `core/grid/spatial_hash.py`  
**Root cause:** `_insert_batch()` (line 144) is a pure Python loop over all points with linear probing.

**Fix strategy:** Extract the inner loop into a `@numba.njit` function that receives individual field arrays (Numba cannot access structured array fields by name inside JIT). The structured `self.cells` array is kept unchanged — Numba sees views of each field separately.

```python
import numba

@numba.njit(cache=True)
def _insert_batch_numba(
    ix_arr, iy_arr, ring_id, z_arr, sem_arr,
    # cell field views (all same length = capacity):
    cell_ix, cell_iy, cell_ring, cell_occ, cell_sem,
    cell_cnt, cell_mean_z, cell_m2_z, cell_min_z, cell_max_z,
    cell_overhang, cell_clearance,
    capacity, max_probe_steps,
):
    """Numba-compiled linear-probing hash insert for a single ring's points."""
    active_count_delta = 0
    n = len(ix_arr)
    for i in range(n):
        ix = ix_arr[i]
        iy = iy_arr[i]
        z_val = z_arr[i]
        sem_val = sem_arr[i]

        # Hash: Wang-style mixing
        ux = (ix & 0xFFFF) * 0x1F1F1F1F
        uy = (iy & 0xFFFF) * 0x5F5F5F5F
        ur = (ring_id & 0xFF) * 0x9E3779B9
        h = (ux ^ uy ^ ur) & 0xFFFFFFFF
        h ^= (h >> 16)
        h = (h * 0x85EBCA6B) & 0xFFFFFFFF
        h ^= (h >> 13)
        base_slot = h % capacity

        for step in range(max_probe_steps):
            slot = (base_slot + step) % capacity
            if cell_occ[slot] == 0:
                # Claim empty slot
                cell_ix[slot] = ix
                cell_iy[slot] = iy
                cell_ring[slot] = ring_id
                cell_occ[slot] = 1
                cell_sem[slot] = sem_val
                cell_cnt[slot] = 1
                cell_mean_z[slot] = z_val
                cell_m2_z[slot] = 0.0
                cell_min_z[slot] = z_val
                cell_max_z[slot] = z_val
                cell_overhang[slot] = 999.0
                cell_clearance[slot] = 999.0
                active_count_delta += 1
                break
            elif cell_ix[slot] == ix and cell_iy[slot] == iy and cell_ring[slot] == ring_id:
                # Welford online update
                cnt = cell_cnt[slot]
                mean_old = cell_mean_z[slot]
                delta = z_val - mean_old
                new_cnt = cnt + 1
                new_mean = mean_old + delta / new_cnt
                delta2 = z_val - new_mean
                new_m2 = cell_m2_z[slot] + delta * delta2

                cell_cnt[slot] = min(new_cnt, 255)
                cell_mean_z[slot] = new_mean
                cell_m2_z[slot] = new_m2
                if z_val < cell_min_z[slot]:
                    cell_min_z[slot] = z_val
                if z_val > cell_max_z[slot]:
                    cell_max_z[slot] = z_val

                # Dual-elevation update (simplified inline for Numba):
                max_z = cell_max_z[slot]
                min_z = cell_min_z[slot]
                span = max_z - min_z
                if span > 1.5:  # candidate overhang
                    if z_val > min_z + 0.8:
                        if z_val < cell_overhang[slot]:
                            cell_overhang[slot] = z_val
                        cell_clearance[slot] = cell_overhang[slot] - min_z
                break
    return active_count_delta
```

Then in `SpatialHashGrid._insert_batch()`, replace the Python loop:

```python
def _insert_batch(self, ix_arr, iy_arr, ring_id, z_arr, sem_arr):
    if sem_arr is None:
        sem_arr = np.zeros(len(ix_arr), dtype=np.uint8)

    delta = _insert_batch_numba(
        ix_arr.astype(np.int32), iy_arr.astype(np.int32),
        np.int32(ring_id), z_arr.astype(np.float64),
        sem_arr.astype(np.uint8),
        self.cells["ix"], self.cells["iy"], self.cells["ring_id"],
        self.cells["occupied"], self.cells["sem_id"],
        self.cells["count"], self.cells["mean_z"], self.cells["m2_z"],
        self.cells["min_z"], self.cells["max_z"],
        self.cells["overhang_z"], self.cells["clearance"],
        np.int64(self.capacity), np.int32(self.max_probe_steps),
    )
    self.active_count += int(delta)
```

**IMPORTANT:**
- The Numba kernel inlines Welford update (to avoid function calls inside JIT).
- The Numba kernel inlines a simplified dual-elevation update. The Python `DualElevationExtractor` is preserved for unit tests (which call it directly) and for the Python fallback path.
- `cache=True` on `@numba.njit` means the cold compile only happens once per version.
- The first call will be slow (~1-3s JIT compile). This **must** be measured separately as a "cold" run and excluded from sustained FPS numbers.
